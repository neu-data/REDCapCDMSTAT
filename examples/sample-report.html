Longitudinal trends

num <- intersect(c("sbp", "weight", "hba1c"), nx$metadata$field_name)
if (!length(num)) num <- head(baseline_vars(nx$metadata), 1)
op <- par(mfrow = c(1, length(num)), mar = c(4, 4, 2, 1))
for (f in num) {
  tr <- nx$trend(f)
  groups <- unique(tr$group)
  ev <- levels(tr$event)
  plot(NA, xlim = c(1, length(ev)), ylim = range(c(tr$lower, tr$upper), na.rm = TRUE), xaxt = "n",
       xlab = "", ylab = paste("Mean", f, "(95% CI)"), main = f)
  axis(1, seq_along(ev), sub("_arm_[0-9]+$", "", ev))
  for (k in seq_along(groups)) {
    s <- tr[tr$group == groups[k], ]
    x <- as.integer(s$event) + (k - 1.5) * 0.08
    lines(x, s$mean, col = k, lwd = 2, type = "b", pch = 19)
    arrows(x, s$lower, x, s$upper, angle = 90, code = 3, length = 0.03, col = k)
  }
  legend("topright", groups, col = seq_along(groups), lwd = 2, bty = "n", cex = .8)
}


par(op)
